cmake_minimum_required(VERSION 3.20)
project(smr_project LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_library(smr src/harris_michael.cpp)
target_include_directories(smr PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(smr PUBLIC Threads::Threads)
if(MSVC)
  target_compile_options(smr PRIVATE /W4)
else()
  target_compile_options(smr PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(test_list tests/test_list.cpp)
target_link_libraries(test_list PRIVATE smr)
add_executable(benchmark benchmarks/benchmark.cpp)
target_link_libraries(benchmark PRIVATE smr)
enable_testing()
add_test(NAME list_test COMMAND test_list)
